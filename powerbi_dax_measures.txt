-- POWER BI DAX MEASURES

Total Spend =
SUM(Transactions[transaction_amount])

Total Transactions =
COUNTROWS(Transactions)

Total Customers =
DISTINCTCOUNT(Customers[customer_id])

Average Transaction Value =
DIVIDE([Total Spend], [Total Transactions])

Spend Per Customer =
DIVIDE([Total Spend], [Total Customers])

Average Credit Limit =
AVERAGE(Customers[credit_limit])

Average Utilization % =
AVERAGE(Customers[utilization_ratio])

Active Customers =
CALCULATE(
    DISTINCTCOUNT(Customers[customer_id]),
    Customers[customer_status] = "Active"
)

Inactive Customers =
CALCULATE(
    DISTINCTCOUNT(Customers[customer_id]),
    Customers[customer_status] = "Inactive"
)

Active Customer % =
DIVIDE([Active Customers], [Total Customers])

High Risk Customers =
CALCULATE(
    DISTINCTCOUNT(Customers[customer_id]),
    Customers[churn_risk_segment] = "High"
)

High Risk Customer % =
DIVIDE([High Risk Customers], [Total Customers])

High Value Customers =
CALCULATE(
    DISTINCTCOUNT(Customers[customer_id]),
    Customers[customer_value_segment] = "High Value"
)

High Value Customer % =
DIVIDE([High Value Customers], [Total Customers])

High Utilization Customers =
CALCULATE(
    DISTINCTCOUNT(Customers[customer_id]),
    Customers[utilization_ratio] >= 0.70
)

High Value Spend =
CALCULATE(
    [Total Spend],
    Customers[customer_value_segment] = "High Value"
)

High Value Spend % =
DIVIDE([High Value Spend], [Total Spend])

Previous Month Spend =
CALCULATE(
    [Total Spend],
    DATEADD('Date'[Date], -1, MONTH)
)

MoM Growth % =
DIVIDE(
    [Total Spend] - [Previous Month Spend],
    [Previous Month Spend]
)

Previous Year Spend =
CALCULATE(
    [Total Spend],
    SAMEPERIODLASTYEAR('Date'[Date])
)

YoY Growth % =
DIVIDE(
    [Total Spend] - [Previous Year Spend],
    [Previous Year Spend]
)

YTD Spend =
TOTALYTD(
    [Total Spend],
    'Date'[Date]
)

Transaction Share % =
DIVIDE(
    [Total Spend],
    CALCULATE(
        [Total Spend],
        ALL(Transactions[transaction_category])
    )
)
